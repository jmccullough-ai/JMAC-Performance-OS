function ATHENA_applyHomeValidations_() {
  const ss = ATHENA_getSpreadsheet_();
  const home = ss.getSheetByName(ATHENA_PRODUCTION.sheets.home);
  const lookup = ss.getSheetByName(ATHENA_PRODUCTION.sheets.lookup);

  const categoryRows = ATHENA_getLookupValuesByCategory_(lookup);

  ATHENA_setDropdown_(home.getRange('B5'), categoryRows['Sport']);
  ATHENA_setDropdown_(home.getRange('B6'), categoryRows['Age Group']);
  ATHENA_setDropdown_(home.getRange('B7'), categoryRows['Experience']);
  ATHENA_setDropdown_(home.getRange('B8'), categoryRows['Phase']);
  ATHENA_setDropdown_(home.getRange('B9'), categoryRows['Training Days']);
  ATHENA_setDropdown_(home.getRange('B10'), categoryRows['Primary Goal']);

  home.getRange('B5').setValue('General Athlete');
  home.getRange('B6').setValue('High School');
  home.getRange('B7').setValue('Beginner');
  home.getRange('B8').setValue('Foundation');
  home.getRange('B9').setValue('3');
  home.getRange('B10').setValue('General Performance');
}

function ATHENA_getLookupValuesByCategory_(lookupSheet) {
  const lastRow = lookupSheet.getLastRow();
  const data = lookupSheet.getRange(2, 1, Math.max(lastRow - 1, 1), 4).getValues();
  const output = {};

  data.forEach(function(row) {
    const category = row[0];
    const value = row[1];
    const active = row[3];
    if (!category || !value || active !== true) return;
    if (!output[category]) output[category] = [];
    output[category].push(value);
  });

  return output;
}

function ATHENA_setDropdown_(range, values) {
  if (!values || values.length === 0) return;
  const rule = SpreadsheetApp.newDataValidation()
    .requireValueInList(values, true)
    .setAllowInvalid(false)
    .build();
  range.setDataValidation(rule);
}
