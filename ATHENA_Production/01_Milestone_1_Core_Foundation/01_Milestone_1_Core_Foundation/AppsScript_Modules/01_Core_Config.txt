function ATHENA_getConfig_() {
  return ATHENA_PRODUCTION;
}

function ATHENA_requiredSheetNames_() {
  const cfg = ATHENA_getConfig_();
  return [
    cfg.sheets.home,
    cfg.sheets.print,
    cfg.sheets.lookup,
    cfg.sheets.sportProfiles,
    cfg.sheets.exerciseMaster,
    cfg.sheets.programRules,
    cfg.sheets.healthCheck
  ];
}

function ATHENA_getSpreadsheet_() {
  return SpreadsheetApp.getActiveSpreadsheet();
}

function ATHENA_toast_(message, title) {
  const ss = ATHENA_getSpreadsheet_();
  ss.toast(message, title || ATHENA_PRODUCTION.systemName, 5);
}
