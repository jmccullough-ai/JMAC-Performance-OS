function ATHENA_refreshWeeklySchedule(){const ss=ATHENA_getSpreadsheet_(),home=ss.getSheetByName(ATHENA_PRODUCTION.sheets.home),schedule=ss.getSheetByName(ATHENA_PRODUCTION.sheets.weeklySchedule);if(!home||!schedule)return;const cal=home.getRange(21,1,7,7).getValues(),train=cal.map(r=>r[1]===true),count=train.filter(Boolean).length,pattern=ATHENA_getSchedulePattern_(train);const out=cal.map((r,i)=>{const day=r[0],is=train[i],prev=i>0?train[i-1]:false,next=i<6?train[i+1]:false;if(!is)return[day,'Off','Off','Off / Recovery','None','None',prev,next,'Off',''];const num=train.slice(0,i+1).filter(Boolean).length,cns=ATHENA_assignCnsType_(count,num,prev,next,pattern),split=ATHENA_assignSplitTheme_(count,num,prev,next,pattern),sp=ATHENA_assignSpeedPowerTheme_(cns,split,num),str=ATHENA_assignStrengthTheme_(split),spacing=ATHENA_assignSpacingLabel_(prev,next);return[day,'Train',cns,split,sp,str,prev,next,spacing,pattern];});schedule.getRange(2,1,7,10).setValues(out);home.getRange('B15').setValue(count);home.getRange('B16').setValue(pattern);for(let i=0;i<7;i++){home.getRange(21+i,3,1,4).setValues([[out[i][2],out[i][3],out[i][4],out[i][5]]]);home.getRange(21+i,7).setValue(out[i][9]);}ATHENA_refreshPrintHeader();ATHENA_toast_('Weekly schedule refreshed.');}
function ATHENA_getSchedulePattern_(train){const count=train.filter(Boolean).length;let b2b=0;for(let i=0;i<train.length-1;i++)if(train[i]&&train[i+1])b2b++;if(count===0)return'No Training Days Selected';if(count===1)return'Single Day';if(b2b===0)return'Spaced';if(b2b>=count-1)return'Compressed / Consecutive';return'Mixed Spacing';}
function ATHENA_assignCnsType_(count,num,prev,next,pattern){if(count===1)return'High CNS';if(pattern==='Spaced'){if(count===2)return'High CNS';if(count===3)return num===2?'Moderate CNS':'High CNS';return num%2===0?'Low CNS':'High CNS';}if(pattern==='Compressed / Consecutive'){if(num===1)return'High CNS';if(num===2)return'Low CNS';return'Moderate CNS';}if(prev&&next)return'Low CNS';if(prev)return'Moderate CNS';return'High CNS';}
function ATHENA_assignSplitTheme_(count,num,prev,next,pattern){if(count===1)return'Total Body Performance';if(count===2)return pattern==='Spaced'?(num===1?'Full Body Acceleration':'Full Body Max Velocity'):(num===1?'Full Body High':'Full Body Low / Recovery Strength');if(count===3){if(pattern==='Spaced')return num===1?'Lower Strength + Acceleration':num===2?'Upper Strength + Max Velocity':'Total Body Power';if(pattern==='Compressed / Consecutive')return num===1?'Lower Strength + Acceleration':num===2?'Upper / Armor Low':'Total Body Power Moderate';return (prev&&next)?'Upper / Armor Low':num===1?'Lower Strength + Acceleration':'Total Body Power';}if(count>=4)return num===1?'Lower Strength + Acceleration':num===2?(prev?'Upper Strength Low / Armor':'Upper Strength + Max Velocity'):num===3?'Lower Power':'Upper Power / Total Body';return'General Performance';}
function ATHENA_assignSpeedPowerTheme_(cns,split,num){if(cns==='Low CNS')return'Movement Prep / Tempo / Low Plyo';if(split.indexOf('Acceleration')!==-1)return'Acceleration / Sled Push';if(split.indexOf('Max Velocity')!==-1)return'Max Velocity / Elastic';if(split.indexOf('Power')!==-1)return'Jumps / Throws / Olympic Derivative';return num%2===1?'Acceleration / Plyo':'Elastic / Med Ball';}
function ATHENA_assignStrengthTheme_(split){if(split.indexOf('Lower')!==-1)return'Lower Strength';if(split.indexOf('Upper')!==-1)return'Upper Strength';if(split.indexOf('Power')!==-1)return'Total Body Power';if(split.indexOf('Low')!==-1)return'Armor / Tissue Capacity';return'Total Body Strength';}
function ATHENA_assignSpacingLabel_(prev,next){if(prev&&next)return'Middle of Back-to-Back';if(prev)return'After Training Day';if(next)return'Before Training Day';return'Rest-Spaced';}
