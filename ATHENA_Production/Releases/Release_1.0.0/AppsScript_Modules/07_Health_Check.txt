function ATHENA_runHealthCheck(){const ss=ATHENA_getSpreadsheet_(),health=ss.getSheetByName(ATHENA_PRODUCTION.sheets.healthCheck)||ss.insertSheet(ATHENA_PRODUCTION.sheets.healthCheck),results=[];ATHENA_requiredSheetNames_().forEach(n=>{const ex=ss.getSheetByName(n)!==null;results.push(['Required sheet exists: '+n,ex?'PASS':'FAIL',ex?'Found':'Missing']);});results.push(ATHENA_checkHeader_(ss,ATHENA_PRODUCTION.sheets.weeklySchedule,['Day','Training_Status','CNS_Type','Split_Theme','Speed_Power_Theme','Strength_Theme','Previous_Day_Training','Next_Day_Training','Spacing_Label','Notes']));results.push(ATHENA_checkHeader_(ss,ATHENA_PRODUCTION.sheets.exerciseMaster,['Exercise_ID','Exercise_Name','Category','Pattern','Plane','Progression_Family','Min_Athlete_Level','Max_Athlete_Level','Primary_Quality','Secondary_Quality','Equipment','CNS_Level','Complexity','High_Low_Type','Gender_Bias','Coach_Tag','Active']));health.clear();ATHENA_buildHealthCheckSheet_(health);health.getRange(5,1,results.length,3).setValues(results).setBorder(true,true,true,true,true,true);results.forEach((r,i)=>health.getRange(5+i,2).setBackground(r[1]==='PASS'?ATHENA_PRODUCTION.colors.green:ATHENA_PRODUCTION.colors.red));const pass=results.filter(r=>r[1]==='PASS').length,fail=results.filter(r=>r[1]==='FAIL').length;health.getRange('A2:H2').setValue('PASS: '+pass+' | FAIL: '+fail+' | Version: '+ATHENA_PRODUCTION.version);ATHENA_toast_('Health Check Complete — PASS: '+pass+' FAIL: '+fail);}
function ATHENA_checkHeader_(ss,sheetName,expected){const sh=ss.getSheetByName(sheetName);if(!sh)return['Header check: '+sheetName,'FAIL','Sheet missing'];const actual=sh.getRange(1,1,1,expected.length).getValues()[0],ok=expected.every((h,i)=>actual[i]===h);return['Header check: '+sheetName,ok?'PASS':'FAIL',ok?'Headers match':'Expected: '+expected.join(', ')];}
